\centerline{\bf \Huge Методы решения неравенств I}

\problem{1}
\textbf{Метод Штурма.}
Для положительных чисел $a, b$ и $c$ с суммой 1 докажите, что

$$
4(a b+b c+a c)-9 a b c \leqslant 1 .
$$

\problem{2}
\textbf{Гомогенизация.}
Про положительные числа $a, b, c$ известно, что $\frac{3}{a b c} \geqslant a+b+c$. Докажите, что выполнено неравенство

$$
\frac{1}{a}+\frac{1}{b}+\frac{1}{c} \geqslant a+b+c .
$$

\problem{3}
\textbf{Линейный движ.}
Докажите, что для любых действительных чисел $a, b, c, d$ выполнено неравенство

$$
\left(a^2-a+1\right)\left(b^2-b+1\right)\left(c^2-c+1\right)\left(d^2-d+1\right) \geqslant \frac{9}{16}(a-b)(b-c)(c-d)(d-a) .
$$

\problem{4}
\textbf{Выпуклость-впуклость.}
Для $a, b, c \in[0,1]$ докажите неравенство

$$
\frac{a}{b+c+1}+\frac{b}{c+a+1}+\frac{c}{a+b+1}+(1-a)(1-b)(1-c) \leqslant 1 .
$$

\problem{5}
\textbf{Транснеравенство/Выравнивание знаменателей.}
Для положительных чисел $a$, $b$ и $c$ докажите неравенство
$$\cfrac a b+\cfrac b c+\cfrac c a\geqslant \cfrac{a+1}{b+1}+\cfrac{b+1}{c+1}+\cfrac{c+1}{a+1}$$

\problem{6}
\textbf{Смена знака.}
Даны неотрицательные числа $a, b, c, d$ такие, что $a+b+c+d=8$. Докажите, что

$$
\frac{a^3}{a^2+b+c}+\frac{b^3}{b^2+c+d}+\frac{c^3}{c^2+d+a}+\frac{d^3}{d^2+a+b} \geqslant 4
$$
