\problem{1}
Решите функциональное уравнение $f(x)+3 f\left(\frac{1}{x}\right)=4 x$.

%\textit{Подсказка: $x \to \dfrac{1}{x}$}

\problem{2}
$\bigl[\textbf{ПВГ, 2015, 9-10 класс}\bigr]$
Найдите функцию $f(x)$, о которой известно, что

$$
f(x)=\left\{\begin{array}{ll}
x \cdot f\left(\frac{2 x+3}{x-2}\right)+3 & \text { при } x \neq 2 \\
0 & \text { при } x=2
\end{array} .\right.
$$

%\textit{Подсказка: $x \to \dfrac{2x+3}{x-2}$}

%\textit{Подсказка: $x = 1, y = 1$}

\problem{4}

\problem{5}

%\textit{Подсказка: $x =0, y = x, x \to \dfrac{x}{2}$}

\problem{6}
Решите функциональное уравнение $f(x)+y^2=f(y)+x^2-y+x$.

%\textit{Подсказка: если для любых $x,y \in \mathbb{R}$ верно, что $g(x) = g(y)$, то $g(x) = const$}

\problem{7}
$\bigl[\textbf{ОММО, 2023}\bigr]$
Найдите все функции $f: \mathbb{R} \rightarrow \mathbb{R}$, для которых существует такое вещественное число $a$, что при всех вещественных $x, y$ выполнено равенство

$$
2 f(x y+3)=f(x) f(y)-f(x)-2 y+a .
$$

%\textit{Подсказка: поменяйте переменные $x$ и $y$ местами.}
