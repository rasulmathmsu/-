В задачах $1-5$ надо придумать два доказательства.

\problem{1}
\textbf{(Бином Ньютона)} $(a+b)^n=C_n^0 a^n+C_n^1 a^{n-1} b+\ldots+C_n^k a^{n-k} b^k+\ldots+C_n^n b^n$.

\problem{2}
$C_n^0+C_n^1+\ldots+C_n^n=2^n$.

\problem{3}
$C_n^0-C_n^1+\ldots+(-1)^n C_n^n=0$.

\problem{4}
$C_n^0 2^0+C_n^1 2^1+\ldots+C_n^n 2^n=3^n$.

\problem{5}
$C_r^m C_m^k=C_r^k C_{r-k}^{m-k}$.

\problem{6}
$C_p^0 C_q^m+C_p^1 C_q^{m-1}+\ldots+C_p^{m-1} C_q^1+C_p^m C_q^0=C_{p+q}^m$.

\problem{7}
$\left(C_n^0\right)^2+\left(C_n^1\right)^2+\ldots+\left(C_n^n\right)^2=C_{2 n}^n$.

\problem{8}
$C_n^k=C_{n-1}^{k-1}+C_{n-2}^{k-1}+\ldots+C_{k-1}^{k-1}$.

\problem{9}
$C_{n-1}^{k-1} C_n^{k+1} C_{n+1}^k=C_{n-1}^k C_{n+1}^{k+1} C_n^{k-1}$.
