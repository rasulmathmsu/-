\centerline{\bf \Huge Неравенства}

\begin{flushright}
        \scriptsize{Одно только делает человека человеком:\\ знание о социальном неравенстве.\\Александр Блок.}

\problem{1!}
Доказать неравенство $\dfrac{a_1}{a_2} + \dfrac{a_2}{a_3} + \ldots + \dfrac{a_n}{a_1} \geq n$, где $a_1, a_2, \ldots, a_n$ --- положительные числа.

\problem{2!}
Докажите, что выполнено неравенство: $ \sqrt{a+1} + \sqrt{2a - 3} + \sqrt{50a - 3} < 12.$

\problem{3!}
Докажите, что для любого натурального $n geq 2$ имеют место неравенства

\[
n \left(\sqrt[n]{n+1} - 1 \right) < 1 + \dfrac{1}{2} + \dfrac{1}{3} + \ldots + \dfrac{1}{n} < n \left(1 - \dfrac{1}{\sqrt[n]{n}} \right) + 1.
\]


\problem{4}
Положительные числа $a, b, c$ таковы, что $\dfrac{1}{a} + \dfrac{1}{b} + \dfrac{1}{c} \leq 1.$ Докажите, что $\dfrac{1}{\sqrt{ab}} + \dfrac{1}{\sqrt{bc}} + \dfrac{1}{\sqrt{ca}} \leq 1$

\problem{5}
Докажите для $x, y, z, t > 7$, что

\[\sqrt{(x-3)(y-4)(z-5)(t-6)} < (x-2)^2 + (y-5)^2 + (z-7)^2 + (t-4)^2.\]

\problem{6}
\textbf{Неравенство Коши-Буняковского-Шварца и его кенты} 
\begin{itemize}
    \item[a)] \textit{КБШ.} Докажите неравенство:
    \[\left( \sum_{i=1}^{k} a_i b_i \right)^2 \leq \left( \sum_{i=1}^{k} a_i^2 \right) \left( \sum_{i=1}^{k} b_i^2 \right).\]
    \item[b)] Докажите с помощью КБШ неравенство 
    \[a_1^2 + \ldots + a_n^2 \geq \dfrac{(a_1 + \ldots + a_n)^2}{n}.\]
    \item[c)] Пусть $x_1, \ldots, x_n$ — положительные числа. Докажите с помощью КБШ неравенство
    \[(x_1 + \ldots + x_n)\left(\dfrac{1}{x_1} + \ldots + \dfrac{1}{x_n}\right) \geq n^2.\]
\end{itemize}
