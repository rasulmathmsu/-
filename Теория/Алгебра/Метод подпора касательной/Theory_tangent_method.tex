\centerline{\bf \Huge Введение в анализ}


Этот файл предназначен для очень поверхностного знакомства с основными аспектами анализа. В конце файла мы ознакомимся с очередным методом решения неравенств.

\textbf{Определение.} Пусть $\left(a_n\right)_{n=1}^{\infty}$ --- некоторая последовательность действительных чисел. Число $A$ называется пределом последовательности $\left(a_n\right)_{n=1}^{\infty}$, если

$$
\forall \varepsilon>0 \ \exists N_{\varepsilon}: \forall n>N_{\varepsilon} \text { выполнено }\left|a_n-A\right|<\varepsilon \text {. }
$$


Обозначение. $\lim _{n \rightarrow \infty} a_n=A$ или <<$a_n \rightarrow A$ при $n \rightarrow \infty$>>.

\textbf{Определение.}
Если у последовательности есть предел, то она называется \textit{сходящейся}. В ином случае \textit{расходящейся}.

\textbf{Примеры}:

\begin{enumerate}
    \item $a_n = \dfrac{n}{n+1}$ сходится
    \item $a_n = (-1)^n$ расходится
    \item $a_n = \dfrac{n^2}{2^n}$ сходится
\end{enumerate}

\textbf{Свойства предела последовательности:}

\begin{enumerate}
    \item Если предел существует, то он единственный
    \item $\lim _{n \rightarrow \infty}\left(a_n+b_n\right)=\lim _{n \rightarrow \infty} a_n+\lim _{n \rightarrow \infty} b_n$.
    \item $\lim _{n \rightarrow \infty}\left(a_n b_n\right)=\lim _{n \rightarrow \infty} a_n \cdot \lim _{n \rightarrow \infty} b_n$
    \item $\lim _{n \rightarrow \infty} \dfrac{a_n}{b_n}=\dfrac{\lim _{n \rightarrow \infty} a_n}{\lim _{n \rightarrow \infty} b_n}$
\end{enumerate}

\textbf{Определение.} \textit{Функцией} $f$, определённой на множестве $X$ со значениями в множестве $Y$, называют «правило» такое, что каждому элементу $x$ из $X$ соответствует элемент $f(x)$, лежащий в $Y$ и притом только один.

\textbf{Определение (Предел функции по Гейне)} 
Значение $A$ называется пределом (предельным значением) функции $f(x)$ в точке $x_0$, если для любой последовательности точек $\left\{x_n\right\}_{n=1}^{\infty}$, сходящейся к $x_0$, но не содержащей $x_0$ в качестве одного из своих элементов (то есть в проколотой окрестности $x_0$ ), последовательность значений функции $\left\{f\left(x_n\right)\right\}_{n=1}^{\infty}$ сходится к $A$.

\textbf{Определение (Предел функции по Коши)}
Значение $A$ называется пределом (предельным значением) функции $f(x)$ в точке $x_0$, если для любого положительного числа $\varepsilon$ можно подобрать соответствующее ему положительное число $\delta=\delta(\varepsilon)$ такое, что для всех аргументов $x$, удовлетворяющих условию $0<\left|x-x_0\right|<\delta$, выполняется неравенство: $0 \leqslant|f(x)-A|<\varepsilon$, то есть $|f(x)-A|<\varepsilon^{[1]}$.

$$
\lim _{x \rightarrow x_0} f(x)=A \Leftrightarrow\left[\forall \varepsilon>0 \quad \exists \ \delta=\delta(\varepsilon)>0 \quad\forall \ x\left(0<\left|x-x_0\right|<\delta\right) \Rightarrow(|f(x)-A|<\varepsilon)\right],
$$

Простыми словами предел по Коши означает следующее: какую бы точность ты ни потребовал для $f(x)$, можно найти не менее хорошую точность для $x$.

Свойства предела функции аналогичны свойствам предела последовательности (ввиду определения по Гейне).

\textbf{Определение.}
Функция $f$ называется непрерывной в точке $a$, если для любой последовательности $x_n$, сходящейся к $a$, последовательность $f\left(x_n\right)$ сходится к $f(a)$.

\textbf{Определение.}
Функция $f$ называется непрерывной в точке $a$, если для любого $\varepsilon>0$ существует $\delta>0$ такое, что при $|a-x|<\delta$ имеем $|f(a)-f(x)|<\varepsilon$.

\textbf{Важные свойства непрерывных функций:}

\begin{enumerate}
    \item Всякая непрерывная на отрезке функция ограничена.
    \item Всякая непрерывная на отрезке функция достигает своего максимума и минимума.
    \item \textbf{Теорема о промежуточном значении.}
    Непрерывная на отрезке $[a, b]$ функция $f$ принимает все значения между $f(a)$ и $f(b)$.
\end{enumerate}

\centerline{\bf \Large Про производную}

\textbf{Определение.}
Производной функции $f$ в точке $a$ называется значение предела $f^{\prime}(a)=\lim _{x \rightarrow a} \frac{f(x)-f(a)}{x-a}$. Если такой предел существует, то функция $f$ называется дифференцируемой в точке $a$.

\textbf{Определение.}
По индукции определим $n$-ю производную: $f^{(n)}(a)=\left(f^{(n-1)}\right)^{\prime}(a)$.

\textbf{Лемма.}
Функция $f$ имеет производную $f^{\prime}(a)$ в точке $a$ тогда и только тогда, когда $f(x)=f(a)+f^{\prime}(a)(x-a)+r(x)$, где остаточный член $r(x)$ «мал» по сравнению с линейной частью, то есть $\lim_{x \rightarrow a} \frac{r(x)}{x-a}=0$.

\textit{Следствие.}
Если функция $f$ дифференцируема в точке $a$, то $f$ непрерывна в $a$.

\textbf{Лемма.}
Пусть функции $f$ и $g$ дифференцируемы в точке $a$. Тогда функции $f+g$ и $f g$ тоже дифференцируемы в $a$, причём $(f+g)^{\prime}(a)=f^{\prime}(a)+g^{\prime}(a)$ и $(f g)^{\prime}(a)=f(a) g^{\prime}(a)+f^{\prime}(a) g(a)$.

\textbf{Доказательство:}

$$(f+g)^{\prime}(a) = \lim_{x \to a}\frac{(f+g)(x)-(f+g)(a)}{x-a}=\lim_{x \to a}\frac{f(x)+g(x)-f(a)-g(a)}{x-a}=$$

$$=\lim_{x \to a}\frac{f(x)-f(a)}{x-a}+\lim_{x \to a}\frac{g(x)-g(a)}{x-a} = f'(x) + g'(x)$$.

$$(fg)^{\prime}(a) = \lim_{x \to a} \frac{f(x) g(x)-f(a) g(a)}{x-a} = \lim_{x \to a} \frac{g(x)(f(x)-f(a))}{x-a}+ \lim_{x \to a} \frac{f(a)(g(x)-g(a))}{x-a} =$$

$$=\lim_{x \to a} g(x) \frac{f(x)-f(a)}{x-a}+ \lim_{x \to a} f(a) \frac{g(x)-g(a)}{x-a} = f(a) g^{\prime}(a)+f^{\prime}(a) g(a)..$$


\textbf{Лемма.}
Пусть функции $g$ и $f$ дифференцируемы в точках $a$ и $g(a)$ соответственно. Тогда функция $f(g(x))$ дифференцируема в $a$, причём $(f(g(a)))^{\prime}=f^{\prime}(g(a)) g^{\prime}(a)$.

\textbf{Доказательство:}

Рассмотрим композицию $h(x)=f(g(x))$. Нужно доказать существование предела

$$
h^{\prime}(a)=\lim _{x \rightarrow a} \frac{f(g(x))-f(g(a))}{x-a}
$$

и вычислить его.

$$\lim _{x \rightarrow a} \frac{f(g(x))-f(g(a))}{x-a} =\lim_{x\rightarrow a} \frac{f(g(x))-f(g(a))}{g(x)-g(a)} \cdot \frac{g(x)- g(a)}{x-a}  =f^{\prime}(g(a)) g^{\prime}(a) $$


\textit{Следствие.}
Пусть функции $f$ и $g$ дифференцируемы в точке $a$ и $f^{\prime}(a)$ не равно 0 . Тогда $f / g$ дифференцируема в точке $a$ и $(f / g)^{\prime}(a)=\frac{f^{\prime}(a) g(a)-f(a) g^{\prime}(a)}{g(a)^2}$.

\textbf{Определение.}
Точка $x_0$ называется точкой локального максимума (минимума) функции $f$, если существует некоторая окрестность $U\left(x_0\right)$ точки $x_0$ такая, что $f(x) \leqslant \leqslant f\left(x_0\right)\left(f(x) \geqslant f\left(x_0\right)\right)$ для любого $x \in U\left(x_0\right)$.

\textbf{Определение.}
Локальным экстремумом функции $f$ называют все ее локальные минимумы и максимумы.

\textbf{Лемма Ферма.}
Производная дифференцируемой функции в точке локального экстремума равна нулю.

\textbf{Доказательство:}

Пусть не умаляя общности $x_0$ точка локального максимум (для минимума будет аналогично). Рассмотрим пределы слева и справа выражения $\frac{f(x)-f\left(x_0\right)}{x-x_0}$. Если стремиться слева, то предел будет неположителен, а если стремиться справа, то неотрицателен. Ввиду того что функция дифференцируема в этой точке, предел слева и справа должен совпадать. Стало быть он равен нулю.

\textit{Замечание.} Обратное неверно! Например, у функции $f(x)=x^3$ производная в точке $x=0$ равна 0 , но эта точка не является локальным экстремумом.

\textbf{Теорема Ролля или о корне производной.}
Пусть функция $f$ непрерывна на $[a, b]$ и дифференцируема на $(a, b)$, причём $f(a)=f(b)$. Тогда найдётся точка $c \in(a, b)$ такая, что $f^{\prime}(c)=0$.

\textbf{Доказательство:}

Так как $f$ непрерывна на замкнутом отрезке $[a, b]$, то она достигает на нём своих наибольшего и наименьшего значений: существуют точки

$$
x_{\max }, x_{\min } \in[a, b]
$$


такие, что

$$
f\left(x_{\max }\right)=\max _{x \in[a, b]} f(x), \quad f\left(x_{\min }\right)=\min _{x \in[a, b]} f(x) .
$$

Если $f$ постоянна на $[a, b]$, то $f^{\prime}(x)=0$ для всех $x \in(a, b)$, и можно взять любое $c \in(a, b)$.
Пусть теперь $f$ не постоянна. Тогда $\max f>\min f$. Но так как $f(a)=f(b)$, невозможно, чтобы и максимум, и минимум достигались только на концах $a$ и $b$ : иначе оба конца имели бы одновременно и значение максимума, и значение минимума, что означало бы $\max f=\min f$, то есть $f$ постоянна - противоречие.

Следовательно, хотя бы один из экстремумов достигается в некоторой точке $c \in(a, b)$. То есть существует $c \in(a, b)$, такая что $c$ - точка локального максимума или локального минимума функции $f$.

Так как $f$ дифференцируема в каждой точке $(a, b)$, в частности дифференцируема в найденной точке $c$. По лемме Ферма производная в точке локального экстремума равна нулю, значит $f^{\prime}(c)=0.$ Точка $c \in(a, b)$ найдена.

\textbf{Теорема Лагранжа.}
Пусть функция $f$ непрерывна на $[a, b]$ и дифференцируема на $(a, b)$. Тогда найдётся точка $c \in(a, b)$ такая, что $f^{\prime}(c)=\frac{f(b)-f(a)}{b-a}$.

\newpage

\hspace{3mm}

\textbf{Доказательство:}

Рассмотрим функцию прямой, проходящей через точки $(a, f(a))$ и $(b, f(b))$ :

$$
\ell(x)=f(a)+\frac{f(b)-f(a)}{b-a}(x-a) .
$$


Определим новую функцию

$$
\varphi(x)=f(x)-\ell(x) .
$$


Тогда:

1. $\varphi$ непрерывна на $[a, b]$ (как разность непрерывных функций) и дифференцируема на $(a, b)$ (как разность дифференцируемых функций).

2. Посчитаем значения на концах:

$$
\begin{gathered}
\varphi(a)=f(a)-\ell(a)=f(a)-f(a)=0, \\
\varphi(b)=f(b)-\ell(b)=f(b)-\left(f(a)+\frac{f(b)-f(a)}{b-a}(b-a)\right)=f(b)-f(b)=0 .
\end{gathered}
$$


То есть $\varphi(a)=\varphi(b)$.

Значит, к $\varphi$ применима теорема Ролля: существует $c \in(a, b)$ такое, что $\varphi^{\prime}(c)=0.$ Ho

$$
\varphi^{\prime}(x)=f^{\prime}(x)-\ell^{\prime}(x)=f^{\prime}(x)-\frac{f(b)-f(a)}{b-a} .
$$


Поэтому из $\varphi^{\prime}(c)=0$ следует

$$
f^{\prime}(c)-\frac{f(b)-f(a)}{b-a}=0,
$$


Tо есть

$$
f^{\prime}(c)=\frac{f(b)-f(a)}{b-a} .
$$

\textbf{Теорема.}
Пусть функция $f$ непрерывна на $[a, b]$ и дифференцируема на $(a, b)$. Функция $f$ монотонно возрастает тогда и только тогда, когда $f^{\prime}(x) \geqslant 0$ при всех $x \in(a, b)$. Аналогично функция монотонно убывает тогда и только тогда, когда $f^{\prime}(x) \leqslant 0$ при всех $x \in(a, b)$.

\textbf{Доказательство:}

$1) \Longrightarrow$
Если $f$ возрастает, то $f^{\prime}(x) \geq 0$
Пусть $f$ монотонно возрастает на $[a, b]$, и возьмём произвольную точку $x \in(a, b)$.
По определению производной

$$
f^{\prime}(x)=\lim _{t \rightarrow x} \frac{f(t)-f(x)}{t-x} .
$$


Если $t>x$, то из возрастания $f(t) \geq f(x)$, а $t-x>0$, значит

$$
\frac{f(t)-f(x)}{t-x} \geq 0 .
$$


Если $t<x$, то $f(t) \leq f(x)$, а $t-x<0$, и снова

$$
\frac{f(t)-f(x)}{t-x} \geq 0
$$

(числитель $\leq 0$, знаменатель $<0$, дробь $\geq 0$ ).
Итак, во всякой проколотой окрестности $x$ разностное отношение $\geq 0$. Следовательно, и его предел (если существует, а он существует по дифференцируемости) тоже $\geq 0$ :

$$
f^{\prime}(x) \geq 0 .
$$

$2) \Longleftarrow$
Если $f^{\prime}(x) \geq 0$ на $(a, b)$, то $f$ возрастает.
Пусть теперь $f^{\prime}(x) \geq 0 \quad \forall x \in(a, b) .$

Возьмём любые $x_1, x_2 \in[a, b]$ такие, что $x_1<x_2$. Тогда $f$ непрерывна на $\left[x_1, x_2\right]$ и дифференцируема на ( $x_1, x_2$ ), поэтому по теореме Лагранжа найдётся $c \in\left(x_1, x_2\right)$ такое, что

$$
f^{\prime}(c)=\frac{f\left(x_2\right)-f\left(x_1\right)}{x_2-x_1} .
$$


Но $f^{\prime}(c) \geq 0$ и $x_2-x_1>0$, значит

$$
\frac{f\left(x_2\right)-f\left(x_1\right)}{x_2-x_1} \geq 0 \quad \Rightarrow \quad f\left(x_2\right)-f\left(x_1\right) \geq 0 \quad \Rightarrow \quad f\left(x_2\right) \geq f\left(x_1\right) .
$$

Это верно для любых $x_1<x_2$, значит $f$ монотонно возрастает на $[a, b]$.

Теперь попробуем решить следующее неравенство:

Сумма положительных чисел $a, b, c, d$ равна 4. Докажите неравенство

$$
\frac{1}{1+a^2}+\frac{1}{1+b^2}+\frac{1}{1+c^2}+\frac{1}{1+d^2} \geqslant 2 .
$$

Пользуясь <<доказанным>> выше, напишем оценку на каждую дробь:

$$
\frac{1}{1+a^2}+\frac{1}{1+b^2}+\frac{1}{1+c^2}+\frac{1}{1+d^2} \geqslant \frac{2-a + 2 - b + 2 - c + 2 -d}{2} = \frac{8 - (a+b+c+d)}{2} = 2
$$

Магия! Не правда ли? 


Теперь попробуем объяснить, что произошло и опишем новый метод решения неравенств. 

Пусть дана дифференцируемая в некоторой окрестности функция $f(x)$ и нужно оценить сумму $f(x_1) + f(x_2) + \ldots + f(x_n)$ при некотором линейном ограничении $x_1+x_2+\ldots +x_n = const.$ Очевидно, что нам нужно как-то оценить функцию через линейные множители и именно тут нам поможет касательная. Нам нужно найти такую касательную, которая будет <<подпирать>> сверху или снизу нашу функции, как это было показано на графике выше. Делается это для того чтобы мы могли написать неравенство вида $f(x) \geqslant kx+b$ в случае, где касательная подпирает график снизу и $\leqslant$ в случае когда сверху. Касательная графику дифференцируемой функции существует в каждой точке её графика, поэтому перед нами стоит выбор --- а в какой точке собственно проводить касательную. Выбор очевиден --- касательная должна быть в той точке, в которой достигается равенство в исходном неравенстве (так как такая прямая будет приближать <<наилучшим образом>>), зачастую это просто <<среднее>> значение переменных. 

Хорошо, с точкой разобрались --- как теперь через неё провести касательную? Есть замечательная формула, которую вы прекрасно докажете самостоятельно:

$$
y = f\left(x_0\right)+f^{\prime}\left(x_0\right)\left(x-x_0\right)
$$

уравнение прямой, касающейся функции $f(x)$ в точке $x_0$.

Приведём несколько примеров решения задач.

\textit{Пример 1.}
Положительные числа $a, b$ и $c$ таковы, что $a+b+c=3$. Докажите, что

$$
\frac{a}{1+2 b^3}+\frac{b}{1+2 c^3}+\frac{c}{1+2 a^3} \geqslant 1 .
$$

\textbf{Доказательство:}

Видно, что в условии переменные не разделены и нельзя каждое слагаемое принять как значение какой-то функции в определенной точке. Поэтому проигнорируем числители и рассмотрим функцию $f(x)=\frac{1}{1+2 x^3}$. Так как $a+b+c=3$, равенство достигается при $a=b=c=1$. Построим касательную к графику функции в точке $x=1$ :

$$
f(1)=\frac{1}{3}, \quad f^{\prime}(x)=-\frac{6 x^2}{\left(1+2 x^3\right)^2} \Longrightarrow f^{\prime}(1)=-\frac{6}{9}=-\frac{2}{3} .
$$


Уравнение касательной:

$$
y=\frac{1}{3}-\frac{2}{3}(x-1)=1-\frac{2}{3} x .
$$


Докажем, что график лежит выше касательной при $x>0$ :

$$
\frac{1}{1+2 x^3} \geqslant 1-\frac{2}{3} x .
$$


Преобразуем неравенство:

$$
\begin{gathered}
1 \geqslant\left(1+2 x^3\right)\left(1-\frac{2}{3} x\right)=1-\frac{2}{3} x+2 x^3-\frac{4}{3} x^4 \\
\frac{4}{3} x^4-2 x^3+\frac{2}{3} x \geqslant 0 \Longleftrightarrow 2 x^4-3 x^3+x \geqslant 0
\end{gathered}
$$


Разложим на множители, зная, что $x=1$ - корень кратности 2 :

$$
x\left(2 x^3-3 x^2+1\right)=x(x-1)^2(2 x+1) \geqslant 0 .
$$


Неравенство верно для всех $x>0$.

Применим полученную оценку к исходному выражению:

$$
\begin{gathered}
\frac{a}{1+2 b^3}+\frac{b}{1+2 c^3}+\frac{c}{1+2 a^3} \geqslant a\left(1-\frac{2}{3} b\right)+b\left(1-\frac{2}{3} c\right)+c\left(1-\frac{2}{3} a\right)= \\
=(a+b+c)-\frac{2}{3}(a b+b c+c a)
\end{gathered}
$$


Учитывая условие $a+b+c=3$ и неравенство

$$
3(a b+b c+c a) \leqslant(a+b+c)^2=9,
$$


откуда $a b+b c+c a \leqslant 3$, получаем:

$$
\frac{a}{1+2 b^3}+\frac{b}{1+2 c^3}+\frac{c}{1+2 a^3} \geqslant 3-\frac{2}{3} \cdot 3=1 .
$$

Что и требовалось доказать.

\textit{Замечание.} Как видите не всегда метод решает неравенство в один ход, иногда он является лишь одним из шагов!

\textit{Пример 2.} Любопытный читатель задастся вопросом --- а что делать, если условие на переменные не линейное, а, например, квадратичное. Что тогда? Ответ таков: если в случае линейности условия, мы подпирали прямыми (линейными объектами), то в случае квадратичности (да и любой другой степени) условия будем подпирать параболами (или же теми функциями, которые вынудит условие задачи). Следующая задача ровно про это. 

Даны положительные $a, b, c$ такие, что $a^2+b^2+c^2=3$. Докажите, что

$$
\frac{1}{2-a}+\frac{1}{2-b}+\frac{1}{2-c} \geqslant 3
$$

\textbf{Доказательство:}

Рассмотрим функцию $f(x)=\frac{1}{2-x}$. Попробуем подобрать такую квадратичную функцию $g(x)=k x^2+m$, что $f(x) \geqslant g(x)$ при $x \in(0 ; \sqrt{3})$. Иначе говоря, нужно подобрать касающуюся параболу к функции $f(x)$. Если с касательной прямой у нас была явная формула, то что же делать в этом случае? Есть совершенно чёткий критерий касания двух функций в точке $x_0$.

\textbf{Лемма.} Функции $f(x)$ и $g(x)$ касаются в точке $x_0$ тогда и только тогда, когда $f(x_0) = g(x_0)$ и $f'(x_0) = g'(x_0)$.

Доказывать мы это не будем. Заметим что при $a=b=c$ исходное неравенство обращается в равенство. Выберем такие $k$ и $m$, что $f(1)=g(1)$ и $f^{\prime}(1)=g^{\prime}(1)$, то есть $k=m=\frac{1}{2}$. Понятно, что при $x \in(0 ; \sqrt{3})$ неравенство $\frac{1}{2-x} \geqslant \frac{x^2+1}{2}$ равносильно неравенству $x(x-1)^2 \geqslant 0$, которое верно на рассматриваемом отрезке. Таким образом, левая часть неравенства не превосходит $\frac{a^2+b^2+c^2+3}{2}=3$, что и требовалось.

\centerline{\bf \Large Как считать производную?}

Считать производную несложно, главное держать у себя в голове основные правила и табличные значения. 

\textbf{Правила дифференцирования:}

\begin{enumerate}
    \item Производная суммы/разности равна сумме/разности производных. $(f\pm g)' = f'\pm g'$. Например,

    $$
    (\sqrt{x}-3 x+1)^{\prime}=(\sqrt{x})^{\prime}-(3 x)^{\prime}+1^{\prime}=\frac{1}{2 \sqrt{x}}-3+0=\frac{1}{2 \sqrt{x}}-3
    $$
    
    \item Вынос константы. $(const\cdot f(x))' = const \cdot f'(x)$. Например,

    $$
    \begin{gathered}
    (2 \cos x)^{\prime}=2 \cdot(\cos x)^{\prime}=2 \cdot(-\sin x)=-2 \sin x . \\
    \left(-\frac{x^3}{3}\right)^{\prime}=-\frac{1}{3} \cdot\left(x^3\right)^{\prime}=-\frac{1}{3} \cdot 3 x^2=-x^2
    \end{gathered}
    $$
    \item Производная произведения.  $(f g)^{\prime}(x)=f(x) g^{\prime}(x)+f^{\prime}(x) g(x)$. Например,

    $$
    \begin{gathered}
    \left(\left(x^2-3\right) \cdot e^x\right)^{\prime}=\left(x^2-3\right)^{\prime} \cdot e^x+\left(x^2-3\right) \cdot\left(e^x\right)^{\prime}= \\
    =\left(\left(x^2\right)^{\prime}-3^{\prime}\right) \cdot e^x+\left(x^2-3\right) \cdot e^x=2 x \cdot e^x+\left(x^2-3\right) \cdot e^x=e^x\left(x^2+2 x-3\right)
    \end{gathered}
    $$
    
    \item Производная частного. $(\frac{f}{g})^{\prime}(x)=\frac{f^{\prime}(x) g(x)-f(x) g^{\prime}(x)}{g(x)^2}$. Например,

    $$
    \begin{gathered}
    \left(\frac{x+2}{e^x}\right)^{\prime}=\frac{(x+2)^{\prime} \cdot e^x-(x+2) \cdot\left(e^x\right)^{\prime}}{\left(e^x\right)^2}=\frac{1 \cdot e^x-(x+2) \cdot e^x}{e^{2 x}}= \\
    =\frac{e^x \cdot(1-(x+2))}{e^{2 x}}=-\frac{x+1}{e^x}
    \end{gathered}
    $$
    
    \item Производная сложной функции. $(f(g(x)))^{\prime}=f^{\prime}(g(x)) g^{\prime}(x)$. Например,

    $$
    \begin{gathered}
    \left(\sqrt{-x^2+2-6 x}\right)^{\prime}=\frac{1}{2 \sqrt{-x^2+2-6 x}} \cdot\left(-x^2+2-6 x\right)^{\prime}= \\
    =\frac{-\left(x^2\right)^{\prime}+2^{\prime}-(6 x)^{\prime}}{2 \sqrt{-x^2+2-6 x}}=\frac{-2 x-6}{2 \sqrt{-x^2+2-6 x}}=-\frac{x+3}{\sqrt{-x^2+2-6 x}} \\
    \left(\log _7\left(x^2+16 x+100\right)\right)^{\prime}=\frac{1}{\left(x^2+16 x+100\right) \cdot \ln 7} \cdot\left(x^2+16 x+100\right)^{\prime}= \\
    =\frac{\left(x^2\right)^{\prime}+(16 x)^{\prime}+100^{\prime}}{\left(x^2+16 x+100\right) \cdot \ln 7}=\frac{2 x+16}{\left(x^2+16 x+100\right) \cdot \ln 7}
    \end{gathered}
    $$
\end{enumerate}
