\centerline{\bf \Huge Неравенство Сержтанкяна}

\begin{flushright}
        \scriptsize{Все переменные по умолчанию положительные.}

\problem{1}
$$
\dfrac{1}{a+b}+\dfrac{1}{b+c}+\dfrac{1}{c+a} \geqslant \dfrac{9}{2(a+b+c)} .
$$


\problem{2}
$$
\dfrac{a}{b+c}+\dfrac{b}{c+d}+\dfrac{c}{d+a}+\dfrac{d}{a+b} \geqslant 2
$$

\problem{3}
Дано, что $10 x+y=1$. Найдите минимальное значение выражения $x^2+y^2$.

\problem{4}
Пусть $a b c=1$. Докажите, что

$$
\dfrac{1}{a^3(b+c)}+\dfrac{1}{b^3(c+a)}+\dfrac{1}{c^3(a+b)} \geqslant \dfrac{3}{2} .
$$


\problem{5}
О числах $a_1, a_2, \ldots, a_n$ известно, что $a_1+a_2+\cdots+a_n=1$. Докажите, что

$$
\left(a_1+\dfrac{1}{a_1}\right)^2+\left(a_2+\dfrac{1}{a_2}\right)^2+\cdots+\left(a_n+\dfrac{1}{a_n}\right)^2 \geqslant \dfrac{\left(n^2+1\right)^2}{n} .
$$


\problem{6}
Пусть $a+b+c=1$. Докажите, что $a^2+b^2+c^2 \geqslant \frac{1}{3}$.

\problem{7}
Докажите, что $a \sqrt{a^2+c^2}+b \sqrt{b^2+c^2} \leqslant a^2+b^2+c^2$.

\problem{8}
\textit{(неравенство Минковского).} Пользуясь КБШ, докажите, что

$$
\left(a_1+a_2+\ldots+a_n\right) \cdot\left(\frac{1}{a_1}+\frac{1}{a_2}+\ldots+\frac{1}{a_n}\right) \geqslant n^2 .
$$
