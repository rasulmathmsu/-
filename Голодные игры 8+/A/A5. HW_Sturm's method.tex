\problem{1}
Докажите неравенство Коши

$$
\dfrac{x_1+\ldots+x_n}{n} \geq \sqrt[n]{x_1 \ldots x_n}
$$


для всех неотрицательных $x_1, \ldots, x_n$.

\problem{2}
Докажите, что для любых положительных чисел $x_1, x_2, \ldots, x_n$ (где $n \geq 2$ ), удовлетворяющих условию $x_1+x_2+\ldots+x_n=1$, справедливо неравенство

$$
\left(\dfrac{1}{x_1^2}-1\right)\left(\dfrac{1}{x_2^2}-1\right) \ldots\left(\dfrac{1}{x_n^2}-1\right) \geq\left(n^2-1\right)^n .
$$

\problem{3}
\textit{(Неравенство Гюйгенса)} $\left(1+x_1\right)\left(1+x_2\right) \ldots\left(1+x_n\right) \geq(1+ \left.+\sqrt[n]{x_1 x_2 \ldots x_n}\right)^n$ при любом натуральном $n$, где все $x_i$ неотрицательны.

\problem{4}
Докажите, что для любых $x, y, z \geq 0$ таких, что $x+y+z=\frac{1}{2}$, справедливо неравенство

$$
\frac{1-x}{1+x} \cdot \frac{1-y}{1+y} \cdot \frac{1-z}{1+z} \geq \frac{1}{3}
$$

\problem{5} $x_1^2+x_2^2+\ldots+x_n^2 \geq \dfrac{1}{n}$, где $x_1, x_2, \ldots, x_n \geq 0, x_1+x_2+\ldots$. $\ldots+x_n=1$.

\problem{6} 
$\left(1+\dfrac{1}{x_1}\right)\left(1+\dfrac{1}{x_2}\right) \ldots\left(1+\dfrac{1}{x_n}\right) \geq(n+1)^n$, где все $x_i$ положительны и $x_1+x_2+\ldots+x_n=1$.

\problem{7}
$\dfrac{\left(1+x_1\right)\left(1+x_2\right) \ldots\left(1+x_n\right)}{\left(1-x_1\right)\left(1-x_2\right) \ldots\left(1-x_n\right)} \geq\left(\dfrac{n+1}{n-1}\right)^n$, где все $x_i$ положительны и $x_1+x_2+\ldots+x_n=1 \quad(n \geq 2)$.

\problem{8}
$\dfrac{1}{1+x_1}+\dfrac{1}{1+x_2}+\ldots+\dfrac{1}{1+x_n} \geq \dfrac{n^2}{n+x_1+x_2+\ldots+x_n}$,  где все $x_i$ неотрицательны.
